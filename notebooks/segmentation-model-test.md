## Closed-set semantic segmentation (Mask2Former, 150 ADE20K labels)

```python {.marimo}
seg_checkpoint = "facebook/mask2former-swin-small-ade-semantic"
seg_processor = AutoImageProcessor.from_pretrained(seg_checkpoint)
segmentation_model = (
    Mask2FormerForUniversalSegmentation.from_pretrained(seg_checkpoint)
    .to(device)
    .eval()
)
```

```python {.marimo}
_inputs = seg_processor(images=image_loaded, return_tensors="pt").to(device)

with torch.no_grad():
    _outputs = segmentation_model(**_inputs)

# per-pixel class ids at the original size (PIL size is (w, h), so reverse it)
seg_map = seg_processor.post_process_semantic_segmentation(
    _outputs, target_sizes=[image_loaded.size[::-1]]
)[0].cpu()

_ids, _counts = torch.unique(seg_map, return_counts=True)
_id2label = segmentation_model.config.id2label

segmentation_labels = (
    pl.DataFrame(
        {
            "label": [_id2label[i] for i in _ids.tolist()],
            "coverage": (_counts / seg_map.numel()).tolist(),
        }
    )
    .filter(pl.col("coverage") >= 0.005)  # drop tiny noisy regions
    .sort("coverage", descending=True)
)
segmentation_labels
```

## Actual model section

```python {.marimo}
owl_checkpoint = "google/owlv2-base-patch16-ensemble"
owl_processor = Owlv2Processor.from_pretrained(owl_checkpoint)
owl_model = Owlv2ForObjectDetection.from_pretrained(owl_checkpoint).to(device).eval()

owl_threshold = 0.05

#embed vocab + resize images
inputs = owl_processor(text=[lvis_vocab], images=image_loaded, return_tensors="pt").to(device)
with torch.no_grad():
    #logits are predictions for each image patch. we only feed one so [0]
    logits = owl_model(**inputs).logits[0]

# Best label per box
scores, idx = torch.sigmoid(logits).max(dim=-1)

owl_labels = (
    pl.DataFrame({"label": [lvis_vocab[i] for i in idx.tolist()], "score": scores.tolist()})
    .filter(pl.col("score") >= owl_threshold)
    .group_by("label")
    .agg(max_score=pl.col("score").max(), detections=pl.len())
    .sort("max_score", descending=True)
)
owl_labels
```
