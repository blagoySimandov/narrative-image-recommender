dino_pipe = pipeline(task="image-feature-extraction", model="facebook/dinov2-small")

def add_dino_embeddings(
    df: pl.DataFrame, col: str, batch_size: int = 32
) -> pl.DataFrame:
    out = dino_pipe(df[col].to_list(), pool=True, batch_size=batch_size)
    emb = np.asarray(out, dtype=np.float32).reshape(len(df), -1)
    return df.with_columns(pl.Series("dino_embeddings", emb))

### Code below was generated by AI.

```python {.marimo}
def best_budget_path(G, k, budget):
    """Maximize summed node prizes, using exactly k nodes,
    with total edge cost <= budget.

    Intended for small graphs with nonnegative edge costs.
    """
    if k < 1:
        raise ValueError("k must be >= 1")
    if budget < 0:
        raise ValueError("budget must be nonnegative")
    if any(data["weight"] < 0 for _, _, data in G.edges(data=True)):
        raise ValueError("This solver requires nonnegative costs")

    best_path = None
    best_prize = float("-inf")
    best_cost = float("inf")

    def search(path, visited, cost, prize):
        nonlocal best_path, best_prize, best_cost

        if len(path) == k:
            # Break equal-prize ties using lower transition cost.
            if (
                prize > best_prize
                or (prize == best_prize and cost < best_cost)
            ):
                best_path = path.copy()
                best_prize = prize
                best_cost = cost
            return

        last = path[-1]

        for nxt in G.successors(last):
            if nxt in visited:
                continue

            new_cost = cost + float(G[last][nxt]["weight"])
            if new_cost > budget:
                continue

            search(
                path + [nxt],
                visited | {nxt},
                new_cost,
                prize + float(G.nodes[nxt]["aesthetic"]),
            )

    # Free starting image: try every possible start.
    for start in G.nodes:
        search(
            [start],
            {start},
            0.0,
            float(G.nodes[start]["aesthetic"]),
        )

    return {
        "path": best_path,
        "prize": None if best_path is None else best_prize,
        "cost": None if best_path is None else best_cost,
    }
```

def show_path_strip(path, X, G, thumb=400):
    n = len(path)
    fig, axes = plt.subplots(1, n, figsize=(3 * n, 3.8))
    axes = np.atleast_1d(axes)
    for pos, (ax, node) in enumerate(zip(axes, path)):
        img = Image.open(X["path"][node]).convert("RGB")
        img.thumbnail((thumb, thumb))
        ax.imshow(img)
        ax.set_title(f"{pos + 1}. img {node}\naesthetic {G.nodes[node]['aesthetic']:.1f}", fontsize=9)
        ax.axis("off")
    # transition cost written between consecutive images
    for pos, (u, w) in enumerate(zip(path, path[1:])):
        fig.text((pos + 1) / n, 0.06, f"→ {G[u][w]['weight']:.3f} →",
                 ha="center", fontsize=9, color="dimgray")
    fig.suptitle(f"prize {solution['prize']:.1f}   cost {solution['cost']:.3f} / budget {cb:.3f}")
    fig.subplots_adjust(bottom=0.15, top=0.78, wspace=0.05)
    return fig

show_path_strip(path, X, G)
